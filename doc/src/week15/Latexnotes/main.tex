\section{Quantum Neural Networks}

%================================================
\begin{frame}{QNN as Variational Quantum State}
\[
|\psi(x,\theta)\rangle = U(\theta)U(x)|0\rangle
\]

\[
f(x,\theta)=\langle \psi(x,\theta)|O|\psi(x,\theta)\rangle
\]

\begin{itemize}
\item $U(x)$: feature map
\item $U(\theta)$: variational ansatz
\item $O$: observable
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Circuit Representation}
\centering
\begin{quantikz}
\lstick{$\ket{0}$} & \gate{U(x)} & \gate{R_y(\theta_1)} & \ctrl{1} & \qw \\
\lstick{$\ket{0}$} & \gate{U(x)} & \gate{R_y(\theta_2)} & \targ{} & \qw
\end{quantikz}
\end{frame}

%================================================
\begin{frame}{Pauli Expansion of the Ansatz}
\[
U(\theta)=e^{-iH(\theta)}
\]

\[
H(\theta)=\sum_\alpha \theta_\alpha P_\alpha
\]

\begin{itemize}
\item $P_\alpha \in \{I,X,Y,Z\}^{\otimes n}$
\item Equivalent to many-body Hamiltonian
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Feature Map as Hamiltonian Evolution}
\[
U(x)=e^{-iH(x)}, \quad H(x)=\sum_i x_i Z_i + \sum_{ij} x_ix_j Z_iZ_j
\]

\begin{itemize}
\item Data = coupling constants
\item Embedding = time evolution
\end{itemize}
\end{frame}

%================================================
\begin{frame}{BCH Expansion}
\[
U^\dagger O U = O + i[H,O] + \frac{i^2}{2}[H,[H,O]] + \cdots
\]

\begin{itemize}
\item Generates hierarchical correlations
\item Analogous to coupled-cluster expansions
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Effective Operator Structure}
\[
O_{\text{eff}} = U^\dagger O U
\]

\begin{itemize}
\item Encodes nonlinear transformations
\item Depth $\rightarrow$ higher-order commutators
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Gradient Derivation}
\[
\partial_\theta f =
\bra{\psi} i[H_\theta,O] \ket{\psi}
\]

\begin{itemize}
\item Gradients = commutators
\item Physical interpretation: response functions
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Parameter Shift Rule}
\[
\frac{\partial f}{\partial \theta} =
\frac{1}{2}\left[f(\theta+\frac{\pi}{2}) - f(\theta-\frac{\pi}{2})\right]
\]

\begin{itemize}
\item Exact gradient
\item Measurement-based evaluation
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Quantum Fisher Information}
\[
F_{ij} = 4\,\mathrm{Re}\left(
\langle \partial_i\psi|\partial_j\psi\rangle -
\langle \partial_i\psi|\psi\rangle
\langle \psi|\partial_j\psi\rangle
\right)
\]
\end{frame}

%================================================
\begin{frame}{QFI as Metric Tensor}
\[
ds^2 = \sum_{ij} F_{ij} d\theta_i d\theta_j
\]

\begin{itemize}
\item Defines Riemannian geometry
\item Distance between quantum states
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Natural Gradient}
\[
\dot{\theta} = F^{-1} \nabla C
\]

\begin{itemize}
\item Corrects Euclidean gradient
\item Respects quantum geometry
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Time-Dependent Variational Principle}
\[
\delta \| (i\partial_t - H)|\psi\rangle \| = 0
\]
\end{frame}

%================================================
\begin{frame}{TDVP Derivation}
Project Schrödinger equation onto tangent space:

\[
\langle \partial_i\psi | (i\partial_t - H)|\psi\rangle = 0
\]

\[
\sum_j F_{ij}\dot{\theta}_j = C_i
\]

\[
C_i = \mathrm{Im}\langle \partial_i\psi|H|\psi\rangle
\]
\end{frame}

%================================================
\begin{frame}{Interpretation of TDVP}
\begin{itemize}
\item QNN training = classical dynamical system
\item Parameters evolve like generalized coordinates
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Loss Landscape}
\[
C(\theta)=\sum_i (f(x_i,\theta)-y_i)^2
\]

\begin{itemize}
\item Highly non-convex
\item Structured by geometry
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Barren Plateaus}
\[
\mathrm{Var}(\nabla C) \sim \frac{1}{2^n}
\]

\begin{itemize}
\item Gradient vanishes exponentially
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Origin of Barren Plateaus}
\begin{itemize}
\item Random circuits $\rightarrow$ unitary 2-designs
\item Expectation values concentrate
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Avoiding Plateaus}
\begin{itemize}
\item Local cost functions
\item Problem-inspired ansätze
\item Layerwise training
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Entanglement and Expressivity}
\[
S(\rho_A) = -\mathrm{Tr}(\rho_A \log \rho_A)
\]

\begin{itemize}
\item Controls complexity
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Expressivity Regimes}
\begin{itemize}
\item Low entanglement → classical-like
\item Moderate → optimal learning
\item High → concentration of measure
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Quantum Neural Tangent Kernel}

K(x,x') = \sum_i \frac{\partial f(x)}{\partial \theta_i}
\frac{\partial f(x')}{\partial \theta_i}
\]
\end{frame}

%================================================
\begin{frame}{Interpretation of NTK}

\item Connects QNN $\leftrightarrow$ kernel methods
\end{itemize}
\end{frame}

%================================================
\begin{frame}{QNN as Quantum Control}

U(\theta)=\mathcal{T} e^{-i\int H(t)dt}

\begin{itemize}

\item Learning = control optimization
\end{itemize}
\end{frame}

%================================================

\end{itemize}
\end{frame}

\section{Variational Quantum Eigensolver}

%================================================
\begin{frame}{Variational Principle}
Given Hamiltonian $H$:

\[
E_0 = \min_{\psi} \langle \psi | H | \psi \rangle
\]

For trial state $|\psi(\theta)\rangle$:

\[
E(\theta) \ge E_0
\]
\end{frame}

%================================================
\begin{frame}{Variational Ansatz}
\[
|\psi(\theta)\rangle = U(\theta)|0\rangle
\]

\[
U(\theta)=e^{-iH(\theta)}, \quad H(\theta)=\sum_\alpha \theta_\alpha P_\alpha
\]
\end{frame}

%================================================
\begin{frame}{Energy Functional}
\[
E(\theta)=\langle \psi(\theta)|H|\psi(\theta)\rangle
\]

\begin{itemize}
\item Optimization in parameter space
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Hamiltonian Decomposition}
\[
H=\sum_i c_i P_i
\]

\begin{itemize}
\item $P_i$: Pauli strings
\item Measurement reduces to expectation values
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Measurement Strategy}
\[
E(\theta)=\sum_i c_i \langle P_i \rangle
\]

\begin{itemize}
\item Requires repeated measurements
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Gradient Derivation}
\[
\frac{\partial E}{\partial \theta}
=
i \langle \psi | [H_\theta, H] | \psi \rangle
\]

\begin{itemize}
\item Structure similar to QNN gradients
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Parameter Shift Rule}
\[
\frac{\partial E}{\partial \theta} =
\frac{1}{2}\left[E(\theta+\frac{\pi}{2}) - E(\theta-\frac{\pi}{2})\right]
\]
\end{frame}

%================================================
\begin{frame}{Circuit}
\centering
\begin{quantikz}
\lstick{$\ket{0}$} & \gate{U(\theta)} & \meter{}
\end{quantikz}
\end{frame}

%================================================
\begin{frame}{Connection to TDVP}
\[
\sum_j F_{ij} \dot{\theta}_j = C_i
\]

\begin{itemize}
\item VQE optimization = projected dynamics
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Unitary Coupled Cluster (UCC)}
\[
U = e^{T - T^\dagger}
\]

\[
T = \sum_{ai} t_{ai} a_a^\dagger a_i + \cdots
\]

\begin{itemize}
\item Systematic ansatz
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Trotterization}

e^{A+B} \approx e^A e^B
\]

\begin{itemize}

\end{itemize}
\end{frame}

%================================================
\begin{frame}{BCH Expansion}

e^{-T}He^T = H + [H,T] + \frac{1}{2}[[H,T],T] + \cdots

\begin{itemize}
\item Connection to coupled cluster theory
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Expressivity}

\end{frame}

%================================================
\begin{frame}{Noise and Sampling}

\begin{frame}{Summary}

\section{Quantum Approximate Optimization Algorithm}

%================================================
\begin{frame}{Optimization Problem}
Given cost function $C(z)$:

\[
H_C |z\rangle = C(z)|z\rangle
\]
\end{frame}

%================================================
\begin{frame}{QAOA Ansatz}
\[
|\gamma,\beta\rangle = \prod_{l=1}^p e^{-i\beta_l H_M} e^{-i\gamma_l H_C} |+\rangle
\]
\end{frame}

%================================================
\begin{frame}{Mixer Hamiltonian}
\[
H_M = \sum_i X_i
\]
\end{frame}

%================================================
\begin{frame}{Circuit (p=1)}
\centering
\begin{quantikz}
\lstick{$\ket{+}$} & \gate{e^{-i\gamma H_C}} & \gate{e^{-i\beta H_M}} & \qw
\end{quantikz}
\end{frame}

%================================================
\begin{frame}{Expectation Value}
\[
C(\gamma,\beta)=\langle \gamma,\beta|H_C|\gamma,\beta\rangle
\]
\end{frame}

%================================================
\begin{frame}{Connection to Adiabatic Evolution}
\[
H(s) = (1-s)H_M + sH_C
\]

\begin{itemize}
\item QAOA = discretized adiabatic evolution
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Trotter Expansion}
\[
e^{-i(H_M + H_C)t} \approx e^{-iH_M \Delta t} e^{-iH_C \Delta t}
\]
\end{frame}

%================================================
\begin{frame}{MaxCut Hamiltonian}
\[
H_C = \sum_{\langle ij\rangle} \frac{1 - Z_i Z_j}{2}
\]
\end{frame}

%================================================
\begin{frame}{Gradient}
\[
\partial_\gamma C = i \langle [H_C, H_{\text{eff}}] \rangle
\]
\end{frame}

%================================================
\begin{frame}{Landscape}
\begin{itemize}
\item Parameter space $(\gamma,\beta)$
\item Non-convex structure
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Relation to VQE}
\begin{itemize}
\item QAOA = structured ansatz
\item VQE = general ansatz
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Relation to QNN}
\begin{itemize}
\item QAOA is a QNN with specific Hamiltonians
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Expressivity vs Depth}
\begin{itemize}
\item Larger $p$ → better approximation
\item but harder optimization
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Summary}
\begin{itemize}
\item QAOA = variational optimization algorithm
\item derived from adiabatic evolution
\item connects optimization and quantum dynamics
\end{itemize}
\end{frame}

\section{Quantum Support Vector Machines}

%================================================
\begin{frame}{SVM: Primal Problem}
Given data $(x_i,y_i)$ with $y_i \in \{-1,1\}$:

\[
\min_{w,b} \frac{1}{2}\|w\|^2
\]

subject to:
\[
y_i(w \cdot x_i + b) \ge 1
\]
\end{frame}

%================================================
\begin{frame}{Dual Formulation}
Introduce Lagrange multipliers $\alpha_i$:

\[
\max_{\alpha}
\sum_i \alpha_i
-
\frac{1}{2}\sum_{ij} \alpha_i \alpha_j y_i y_j (x_i \cdot x_j)
\]
\end{frame}

%================================================
\begin{frame}{Kernel Trick}
Replace inner product:
\[
x_i \cdot x_j \rightarrow K(x_i,x_j)
\]

Implicit mapping:
\[
x \rightarrow \phi(x)
\]

\[
K(x,x') = \langle \phi(x), \phi(x') \rangle
\]
\end{frame}

%================================================
\begin{frame}{Mercer Theorem}
If $K(x,x')$ is positive definite:

\[
K(x,x') = \sum_k \lambda_k \psi_k(x)\psi_k(x')
\]

\begin{itemize}
\item Defines embedding into RKHS
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Quantum Feature Map}
Define:
\[
|\phi(x)\rangle = U(x)|0\rangle
\]

Then:
\[
K(x,x') = |\langle \phi(x)|\phi(x')\rangle|^2
\]
\end{frame}

%================================================
\begin{frame}{Pauli Expansion of Feature Map}
\[
U(x) = e^{iH(x)}, \quad H(x)=\sum_\alpha f_\alpha(x) P_\alpha
\]

Example:
\[
H(x)=\sum_i x_i Z_i + \sum_{ij} x_i x_j Z_i Z_j
\]

\begin{itemize}
\item Equivalent to spin Hamiltonian
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Kernel as Fidelity}
\[
K(x,x') = |\langle \phi(x)|\phi(x')\rangle|^2
\]

\begin{itemize}
\item Transition probability
\item Quantum overlap
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Swap Test Circuit}
\centering
\begin{quantikz}
\lstick{$\ket{0}$} & \gate{H} & \ctrl{1} & \gate{H} & \meter{} \\
\lstick{$\phi(x)$} & \qw & \swap{1} & \qw & \qw \\
\lstick{$\phi(x')$} & \qw & \targX{}{} & \qw & \qw
\end{quantikz}

\[
P(0) = \frac{1 + |\langle \phi(x)|\phi(x')\rangle|^2}{2}
\]
\end{frame}

%================================================
\begin{frame}{Direct Fidelity Circuit}
Alternative:
\centering
\begin{quantikz}
\lstick{$\ket{0}$} & \gate{U(x)} & \gate{U^\dagger(x')} & \meter{}
\end{quantikz}

\[
P(0) = |\langle \phi(x)|\phi(x')\rangle|^2
\]
\end{frame}

%================================================
\begin{frame}{Kernel Matrix}
\[
K_{ij} = |\langle \phi(x_i)|\phi(x_j)\rangle|^2
\]

\begin{itemize}
\item Evaluated on quantum device
\item Used in classical optimization
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Decision Function}
\[
f(x) = \sum_i \alpha_i y_i K(x_i,x) + b
\]
\end{frame}

%================================================
\begin{frame}{Expressivity of Quantum Kernels}
Kernel determined by:
\[
U^\dagger(x)U(x')
\]

\begin{itemize}
\item Encodes nonlinear correlations
\item Depends on circuit depth and entanglement
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Kernel Expansion}
\[
K(x,x') = |\langle 0| U^\dagger(x)U(x') |0\rangle|^2
\]

\begin{itemize}
\item Equivalent to expectation value
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Expressivity vs Entanglement}
\begin{itemize}
\item No entanglement → separable kernel
\item Moderate entanglement → rich structure
\item High entanglement → concentration of measure
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Concentration of Measure}
\[
|\langle \phi(x)|\phi(x')\rangle|^2 \approx \frac{1}{2^n}
\]

\begin{itemize}
\item Kernel becomes nearly constant
\item Loss of discriminative power
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Quantum Neural Tangent Kernel Connection}
\[
K_{\text{NTK}}(x,x') = \sum_i \frac{\partial f(x)}{\partial \theta_i}
\frac{\partial f(x')}{\partial \theta_i}
\]

\begin{itemize}

\end{itemize}
\end{frame}

%================================================
\begin{frame}{Connection to HHL}

\begin{itemize}
\item HHL can solve this system

\end{frame}

%================================================
\begin{frame}{Physics Interpretation}

\]

\begin{itemize}
\item Overlap of quantum states
\item Analogous to correlation functions
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Hilbert Space Geometry}

\item Classification = geometric separation
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Advantages}

\item Potential quantum advantage

\end{frame}

%================================================
\begin{frame}{Limitations}

\end{frame}

%================================================
\begin{frame}{Summary}

\section{HHL Algorithm}

%================================================
\begin{frame}{Problem Statement}
We wish to solve the linear system:
\[
A x = b
\]

with:
\begin{itemize}
\item $A \in \mathbb{C}^{N \times N}$ Hermitian
\item $|b\rangle$ efficiently preparable
\end{itemize}

Goal:
\[
|x\rangle \propto A^{-1}|b\rangle
\]
\end{frame}

%================================================
\begin{frame}{Spectral Decomposition}
\[
A = \sum_j \lambda_j |u_j\rangle\langle u_j|
\]

Expand input:
\[
|b\rangle = \sum_j \beta_j |u_j\rangle
\]

Then:
\[
A^{-1}|b\rangle = \sum_j \frac{\beta_j}{\lambda_j} |u_j\rangle
\]
\end{frame}

%================================================
\begin{frame}{Key Idea of HHL}
HHL implements:
\[
\lambda_j \;\longrightarrow\; \frac{1}{\lambda_j}
\]

on the eigenvalues of $A$.

\begin{itemize}
\item This is a \textbf{spectral transformation}
\item Achieved via Quantum Phase Estimation (QPE)
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Quantum Phase Estimation (QPE)}
QPE extracts eigenvalues of $A$:

\[
U = e^{iAt}
\]

\[
U|u_j\rangle = e^{i\lambda_j t}|u_j\rangle
\]

QPE maps:
\[
|u_j\rangle|0\rangle \rightarrow |u_j\rangle|\lambda_j\rangle
\]
\end{frame}

%================================================
\begin{frame}{HHL Circuit Structure}
\centering
\begin{quantikz}[row sep=0.3cm]
\lstick{$|b\rangle$} & \qw & \gate{QPE} & \qw & \gate{QPE^\dagger} & \qw \\
\lstick{$|0\rangle$} & \qw & \qw & \gate{\lambda^{-1}} & \qw & \qw
\end{quantikz}

\begin{itemize}
\item QPE encodes eigenvalues
\item Controlled rotation applies inverse
\end{itemize}
\end{frame}

%================================================
\begin{frame}{State After QPE}
After QPE:
\[
\sum_j \beta_j |u_j\rangle |\lambda_j\rangle
\]

Eigenvalues are stored in a register.
\end{frame}

%================================================
\begin{frame}{Controlled Rotation}
Introduce ancilla qubit:

\[
|\lambda_j\rangle|0\rangle \rightarrow
|\lambda_j\rangle\left(
\sqrt{1 - \frac{C^2}{\lambda_j^2}}|0\rangle +
\frac{C}{\lambda_j}|1\rangle
\right)
\]

\begin{itemize}
\item Encodes $\lambda_j^{-1}$
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Post-selection}
Condition on ancilla $|1\rangle$:

\[
|\psi\rangle \propto \sum_j \frac{\beta_j}{\lambda_j} |u_j\rangle
\]

This is the desired solution state.
\end{frame}

%================================================
\begin{frame}{Uncomputation}
Apply inverse QPE:

\[
|\lambda_j\rangle \rightarrow |0\rangle
\]

Final state:
\[
|x\rangle \propto A^{-1}|b\rangle
\]
\end{frame}

%================================================
\begin{frame}{Full Circuit (Conceptual)}
\centering
\begin{quantikz}[row sep=0.3cm]
\lstick{$|b\rangle$} & \qw & \gate{QPE} & \qw & \gate{QPE^\dagger} & \qw \\
\lstick{$|0\rangle$} & \qw & \qw & \gate{R_y(\lambda^{-1})} & \qw & \qw \\
\lstick{$|0\rangle$} & \qw & \qw & \meter{} & \qw & \qw
\end{quantikz}
\end{frame}

%================================================
\begin{frame}{Complexity}
Runtime depends on:
\[
\mathcal{O}\left(\kappa^2 \log N\right)
\]

where:
\begin{itemize}
\item $\kappa$ = condition number
\item $N$ = system size
\end{itemize}

Potential exponential speedup if:
\begin{itemize}
\item sparse matrix
\item efficient state preparation
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Limitations}
\begin{itemize}
\item Output is quantum state, not full vector
\item Requires efficient block encoding
\item Sensitive to condition number
\end{itemize}
\end{frame}

%================================================
\begin{frame}{HHL as Green's Function}
\[
A^{-1} \sim (\omega I - H)^{-1}
\]

This is the \textbf{resolvent operator}.

\begin{itemize}
\item Appears in Green's functions
\item Encodes system response
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Connection to Many-Body Physics}
\[
G(\omega) = (\omega I - H)^{-1}
\]

\begin{itemize}
\item Poles correspond to eigenvalues
\item Residues = spectral weights
\end{itemize}
\end{frame}

%================================================
\begin{frame}{Interpretation}
