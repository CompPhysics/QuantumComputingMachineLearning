\section{Worked Exercises}

\begin{frame}{Exercise 1: Partially entangled two-qubit state}
Consider
\[
\ket{\psi(\theta)}=\cos\theta\,\ket{00}+\sin\theta\,\ket{11},\qquad \theta\in[0,\pi/2].
\]
\textbf{Tasks:}
\begin{enumerate}
\item Compute $\rho_{AB}=\ket{\psi(\theta)}\bra{\psi(\theta)}$.
\item Compute $\rho_A=\Tr_B(\rho_{AB})$.
\item Find eigenvalues of $\rho_A$ and compute
\[
S(\rho_A),\quad S_2(\rho_A),\quad S_\alpha(\rho_A).
\]
\item Identify $\theta$ for which entanglement is maximal.
\end{enumerate}

\medskip
\textbf{Hint:} $\rho_A$ will be diagonal in $\{\ket{0},\ket{1}\}$.
\end{frame}

% ==========================================================
\begin{frame}{Solution 1 (full derivation)}
Start with
\[
\rho_{AB}=\ket{\psi}\bra{\psi}=
\cos^2\theta\,\ket{00}\bra{00}
+\cos\theta\sin\theta\,\ket{00}\bra{11}
+\cos\theta\sin\theta\,\ket{11}\bra{00}
+\sin^2\theta\,\ket{11}\bra{11}.
\]

Use $\Tr_B(\ket{ab}\bra{cd})=\braket{d}{b}\ket{a}\bra{c}$:
\begin{align}
\Tr_B(\ket{00}\bra{00})&=\ket{0}\bra{0},\\
\Tr_B(\ket{00}\bra{11})&=\braket{1}{0}\ket{0}\bra{1}=0,\\
\Tr_B(\ket{11}\bra{00})&=\braket{0}{1}\ket{1}\bra{0}=0,\\
\Tr_B(\ket{11}\bra{11})&=\ket{1}\bra{1}.
\end{align}

Therefore,
\[
\rho_A=\cos^2\theta\,\ket{0}\bra{0}+\sin^2\theta\,\ket{1}\bra{1}
=\begin{pmatrix}
\cos^2\theta & 0\\
0 & \sin^2\theta
\end{pmatrix}.
\]

Eigenvalues: $\lambda_1=\cos^2\theta$, $\lambda_2=\sin^2\theta$.
\end{frame}

% ==========================================================
\begin{frame}{Solution 1: Entropies}
Von Neumann entropy:
\[
S(\rho_A)=-\cos^2\theta\,\log_2(\cos^2\theta)-\sin^2\theta\,\log_2(\sin^2\theta).
\]

Second Rényi entropy:
\[
S_2(\rho_A)=-\log_2\Tr(\rho_A^2)
=-\log_2\left(\cos^4\theta+\sin^4\theta\right).
\]

General Rényi:
\[
S_\alpha(\rho_A)=\frac{1}{1-\alpha}\log_2\left((\cos^2\theta)^\alpha+(\sin^2\theta)^\alpha\right).
\]

\medskip
\textbf{Max entanglement:} at $\theta=\pi/4$,
$\lambda_1=\lambda_2=1/2$ so $S(\rho_A)=1$ bit.
\end{frame}

% ==========================================================
\begin{frame}{Exercise 2: Measurement entropy vs state entropy}
Let the single-qubit state be
\[
\rho = p\ket{0}\bra{0}+(1-p)\ket{1}\bra{1},\qquad p\in[0,1].
\]

\textbf{Tasks:}
\begin{enumerate}
\item Compute $S(\rho)$.
\item Compute measurement entropy $H_{\text{meas}}$ for measuring in the $Z$ basis.
\item Compute $H_{\text{meas}}$ for measuring in the $X$ basis.
\end{enumerate}

\medskip
\textbf{Key point:} $S(\rho)$ is basis-independent, while $H_{\text{meas}}$ depends on the measurement.
\end{frame}

% ==========================================================
\begin{frame}{Solution 2: $Z$ and $X$ measurements}
Since $\rho$ is diagonal with eigenvalues $(p,1-p)$:
\[
S(\rho)=-p\log_2 p-(1-p)\log_2(1-p).
\]

\medskip
\textbf{$Z$ basis measurement:}
\[
p(0)=p,\quad p(1)=1-p
\Rightarrow
H_Z=-p\log_2 p-(1-p)\log_2(1-p)=S(\rho).
\]

\medskip
\textbf{$X$ basis measurement:}
$\ket{\pm}=(\ket{0}\pm\ket{1})/\sqrt{2}$, with projectors $\Pi_\pm=\ket{\pm}\bra{\pm}$.
Then
\[
p(\pm)=\Tr(\rho\,\Pi_\pm)=\frac{1}{2},
\]
so
\[
H_X=1,
\]
independent of $p$.

\medskip
Thus generally $H_{\text{meas}}\ge S(\rho)$, with equality in eigenbasis of $\rho$.
\end{frame}

% ==========================================================
\begin{frame}{Exercise 3: Mutual information for a classical mixture}
Consider a two-qubit mixed state
\[
\rho_{AB}= \frac{1}{2}\ket{00}\bra{00}+\frac{1}{2}\ket{11}\bra{11}.
\]

\textbf{Tasks:}
\begin{enumerate}
\item Compute $\rho_A,\rho_B$.
\item Compute $S(\rho_{AB})$, $S(\rho_A)$, $S(\rho_B)$.
\item Compute $I(A\!:\!B)=S(\rho_A)+S(\rho_B)-S(\rho_{AB})$.
\item Compare to a Bell state, where $I(A\!:\!B)=2$.
\end{enumerate}
\end{frame}

% ==========================================================
\begin{frame}{Solution 3: Classical vs quantum correlations}
Compute reduced states:
\[
\rho_A=\rho_B=\frac{I_2}{2}\quad\Rightarrow\quad S(\rho_A)=S(\rho_B)=1.
\]

The global state has eigenvalues $\{1/2,1/2,0,0\}$, hence
\[
S(\rho_{AB})=1.
\]

Therefore
\[
I(A\!:\!B)=1+1-1=1.
\]

\medskip
\textbf{Interpretation:}
\begin{itemize}
\item This state has correlations, but they are purely classical.
\item Bell states have maximal mutual information $I=2$, reflecting strong quantum correlations.
\end{itemize}
\end{frame}

\section{Numerical Examples (Pure Python)}

\begin{frame}{Numerics: what we compute}
We provide \textbf{standard-library-only} Python code to:
\begin{itemize}
\item build two-qubit states/density matrices,
\item compute partial traces,
\item compute eigenvalues (2x2 analytic and 4x4 via a simple power/QR routine),
\item compute von Neumann and Rényi entropies,
\item compute measurement outcome distributions and entropies.
\end{itemize}

\medskip
For teaching, we keep the implementation explicit and transparent.
\end{frame}

% ==========================================================
\begin{frame}[fragile]{Code 1: utilities (partial trace, eigenvalues, entropies)}
\begin{lstlisting}[language=Python]
import math
import cmath

# --- basic helpers ---
def outer(v):
    n = len(v)
    return [[v[i]*v[j].conjugate() for j in range(n)] for i in range(n)]

def partial_trace_B(rho):
    """Tr_B for 2 qubits. Basis: |00>,|01>,|10>,|11>."""
    rhoA = [[0j, 0j],[0j, 0j]]
    for a in (0,1):
        for c in (0,1):
            s = 0j
            for b in (0,1):
                i = 2*a + b
                j = 2*c + b
                s += rho[i][j]
            rhoA[a][c] = s
    return rhoA

def eigvals_2x2(M):
    """Eigenvalues of a 2x2 matrix."""
    a,b = M[0]
    c,d = M[1]
    tr = a + d
    det = a*d - b*c
    disc = tr*tr - 4*det
    root = cmath.sqrt(disc)
    lam1 = 0.5*(tr + root)
    lam2 = 0.5*(tr - root)
    return [lam1.real, lam2.real]

def vn_entropy_from_eigs(lams, eps=1e-15):
    S = 0.0
    for lam in lams:
        lam = max(lam, 0.0)
        if lam > eps:
            S -= lam * (math.log(lam, 2))
    return S

def renyi_entropy_from_eigs(lams, alpha, eps=1e-15):
    if abs(alpha - 1.0) < 1e-12:
        return vn_entropy_from_eigs(lams, eps=eps)
    s = 0.0
    for lam in lams:
        lam = max(lam, 0.0)
        s += lam**alpha
    return (1.0/(1.0-alpha)) * math.log(max(s, eps), 2)
\end{lstlisting}
\end{frame}

import math

def psi_theta(theta):
    # |psi> = cos(theta)|00> + sin(theta)|11>
    return [math.cos(theta), 0j, 0j, math.sin(theta)]

def scan_theta(n=11):
    print("theta   S_vN(rhoA)   S2(rhoA)")
    for k in range(n):
        theta = (0.5*math.pi) * k/(n-1)
        psi = psi_theta(theta)
        rho = outer(psi)
        rhoA = partial_trace_B(rho)
        lams = eigvals_2x2(rhoA)
        SvN = vn_entropy_from_eigs(lams)
        S2  = renyi_entropy_from_eigs(lams, alpha=2.0)
        print(f"{theta:6.3f}   {SvN:10.6f}   {S2:10.6f}")

if __name__ == "__main__":
    scan_theta(n=13)
\end{lstlisting}

\begin{frame}[fragile]{Code 3: Measurement entropy in different bases (single qubit)}
\begin{lstlisting}[language=Python]
import math

def shannon_entropy(ps, eps=1e-15):
    H = 0.0
    for p in ps:
        p = max(p, 0.0)
        if p > eps:
            H -= p*math.log(p, 2)
    return H

def meas_entropy_Z(p):
    # rho = p|0><0| + (1-p)|1><1|
    return shannon_entropy([p, 1-p])

def meas_entropy_X(p):
    # measuring diagonal rho in X basis gives uniform outcomes
    return 1.0

def demo(p_list=(0.1, 0.25, 0.5, 0.9)):
    print("p     H_Z      H_X")
    for p in p_list:
        print(f"{p:3.2f}  {meas_entropy_Z(p):7.5f}  {meas_entropy_X(p):7.5f}")

if __name__ == "__main__":
    demo()
\end{lstlisting}

\section{Open Systems: Lindblad Dynamics and Entropy Production}

\begin{frame}{Open quantum systems: why entropy changes}
Closed (unitary) evolution:
\[
\rho(t)=U(t)\rho(0)U^\dagger(t)\quad \Rightarrow\quad S(\rho(t))=S(\rho(0)).
\]

\medskip
Open system evolution includes environment coupling:
\begin{itemize}
\item decoherence (loss of phase coherence),
\item dissipation (energy exchange),
\item noise-induced mixing.
\end{itemize}

\medskip
As a result, the system entropy can increase (or change non-monotonically).
\end{frame}

% ==========================================================
\begin{frame}{Lindblad master equation (GKSL form)}
Markovian open-system dynamics are described by the Lindblad equation:
\[
\dot{\rho} = -i[H,\rho] + \sum_k \mathcal{D}[L_k]\rho,
\]
where the dissipator is
\[
\mathcal{D}[L]\rho = L\rho L^\dagger - \frac{1}{2}\{L^\dagger L,\rho\}.
\]

\medskip
\textbf{Interpretation:}
\begin{itemize}
\item $H$ generates coherent unitary dynamics.
\item $L_k$ are jump operators describing noise channels.
\end{itemize}
\end{frame}

% ==========================================================
\begin{frame}{Example channels for a single qubit}
\textbf{Pure dephasing} (phase damping):
\[
L=\sqrt{\gamma_\phi}\,\sigma_z.
\]
\begin{itemize}
\item populations unchanged,
\item off-diagonal coherence decays.
\end{itemize}

\medskip
\textbf{Amplitude damping} (energy relaxation):
\[
L=\sqrt{\gamma_1}\,\sigma_-,
\qquad
\sigma_-=\ket{0}\bra{1}.
\]
\begin{itemize}
\item excited state decays to ground state,
\item entropy can increase transiently then decrease as state relaxes.
\end{itemize}
\end{frame}

% ==========================================================
\begin{frame}{Entropy production: qualitative picture}
For an open system, von Neumann entropy obeys
\[
\frac{d}{dt}S(\rho(t)) = -\Tr(\dot{\rho}\,\log\rho).
\]

\medskip
Under Lindblad dynamics, $\dot{\rho}$ has two contributions:
\[
\dot{\rho}= -i[H,\rho] + \sum_k \mathcal{D}[L_k]\rho.
\]

\medskip
Since $\Tr([H,\rho]\log\rho)=0$ (trace cyclicity), entropy change is driven by the dissipators:
\[
\frac{d}{dt}S(\rho(t)) = -\sum_k \Tr\big(\mathcal{D}[L_k]\rho \,\log\rho\big).
\]

\medskip
This motivates defining \textbf{entropy production} as the irreversible component associated with dissipation.
\end{frame}

% ==========================================================
\begin{frame}{Spohn inequality (entropy production rate)}
For a Lindblad semigroup with stationary state $\rho_\ast$,
Spohn's inequality implies non-negative entropy production:
\[
\sigma(t) \equiv -\frac{d}{dt}S(\rho(t)\Vert \rho_\ast) \ge 0,
\]
where $S(\rho\Vert\sigma)=\Tr(\rho(\log\rho-\log\sigma))$ is quantum relative entropy.

\medskip
Expanding:
\[
\sigma(t) = \Tr\Big(\dot{\rho}\,(\log\rho_\ast-\log\rho)\Big)\ge 0.
\]

\medskip
Interpretation:
\begin{itemize}
\item $\sigma(t)$ measures irreversibility.
\item It vanishes only at stationarity ($\rho=\rho_\ast$) or under reversible dynamics.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Open-system numerics (pure Python): dephasing a Bell state}
We simulate local dephasing on qubit $A$ of $\ket{\Phi^+}$:
\[
L=\sqrt{\gamma}\,(\sigma_z\otimes I).
\]
This destroys entanglement and increases $S(\rho_{AB})$ from 0.

\medskip
We use explicit Euler integration for clarity (small $\Delta t$).
\end{frame}

\begin{frame}[fragile]{Code 4: Lindblad dephasing (2 qubits, standard library only)}
\begin{lstlisting}[language=Python]
import math, cmath

def matmul(A,B):
    n,m,p = len(A), len(B[0]), len(B)
    return [[sum(A[i][k]*B[k][j] for k in range(p)) for j in range(m)] for i in range(n)]

def matsum(A,B):
    return [[A[i][j]+B[i][j] for j in range(len(A[0]))] for i in range(len(A))]

def matscale(c,A):
    return [[c*A[i][j] for j in range(len(A[0]))] for i in range(len(A))]

def dagger(A):
    return [[A[j][i].conjugate() for j in range(len(A))] for i in range(len(A[0]))]

def commutator(H,rho):
    return matsum(matmul(H,rho), matscale(-1, matmul(rho,H)))

def anticommutator(A,B):
    return matsum(matmul(A,B), matmul(B,A))

def dissipator(L,rho):
    Ld = dagger(L)
    term1 = matmul(matmul(L,rho), Ld)
    LdL = matmul(Ld,L)
    term2 = matscale(0.5, matmul(LdL, rho))
    term3 = matscale(0.5, matmul(rho, LdL))
    return matsum(term1, matscale(-1, matsum(term2, term3)))

def kron(A,B):
    out=[]
    for i in range(len(A)):
        for ii in range(len(B)):
            row=[]
            for j in range(len(A[0])):
                for jj in range(len(B[0])):
                    row.append(A[i][j]*B[ii][jj])
            out.append(row)
    return out

# Pauli matrices
I2 = [[1+0j,0j],[0j,1+0j]]
Z  = [[1+0j,0j],[0j,-1+0j]]

def outer(v):
    n=len(v)
    return [[v[i]*v[j].conjugate() for j in range(n)] for i in range(n)]

def trace(A):
    return sum(A[i][i] for i in range(len(A)))

def partial_trace_B(rho):
    rhoA=[[0j,0j],[0j,0j]]
    for a in (0,1):
        for c in (0,1):
            s=0j
            for b in (0,1):
                i=2*a+b
                j=2*c+b
                s += rho[i][j]
            rhoA[a][c]=s
    return rhoA

def eigvals_2x2(M):
    a,b = M[0]; c,d = M[1]
    tr=a+d; det=a*d-b*c
    disc=tr*tr-4*det
    root=cmath.sqrt(disc)
    return [0.5*(tr+root).real, 0.5*(tr-root).real]

def vn_entropy_2x2(rhoA, eps=1e-15):
    lams=eigvals_2x2(rhoA)
    S=0.0
    for lam in lams:
        lam=max(lam,0.0)
        if lam>eps:
            S -= lam*math.log(lam,2)
    return S

def simulate_dephasing(gamma=1.0, dt=1e-3, steps=5000):
    # Bell state |Phi+> = (|00> + |11>)/sqrt(2)
    inv=1.0/math.sqrt(2.0)
    psi=[inv,0j,0j,inv]
    rho=outer(psi)

    # H = 0 for pure dephasing demo
    H=[[0j]*4 for _ in range(4)]

    # L = sqrt(gamma) (Z ⊗ I)
    L = matscale(math.sqrt(gamma), kron(Z,I2))

    print("t      S(rho_A)")
    for n in range(steps+1):
        t = n*dt
        if n % (steps//10) == 0:
            rhoA = partial_trace_B(rho)
            print(f"{t:6.3f}  {vn_entropy_2x2(rhoA):9.6f}")

        drho = matsum(
            matscale(-1j, commutator(H,rho)),
            dissipator(L,rho)
        )
        rho = matsum(rho, matscale(dt, drho))

    return rho

if __name__ == "__main__":
    simulate_dephasing(gamma=2.0, dt=2e-3, steps=5000)
\end{lstlisting}
\end{frame}

% ==========================================================
\begin{frame}{What to observe in the dephasing demo}
Local dephasing on one qubit of a Bell pair:
\begin{itemize}
\item destroys coherence between $\ket{00}$ and $\ket{11}$ in $\rho_{AB}$,
\item reduces entanglement,
\item keeps local state maximally mixed: $\rho_A \approx I_2/2$ throughout,
\item hence $S(\rho_A)$ stays close to 1.
\end{itemize}

\medskip
To see entanglement loss numerically, one can monitor:
\begin{itemize}
\item global entropy $S(\rho_{AB})$ (requires 4x4 eigenvalues),
\item concurrence / negativity (beyond today's scope),
\item decay of off-diagonal terms $\rho_{00,11}$.
\end{itemize}
\end{frame}

% ==========================================================
\begin{frame}{Optional extension: entropy balance}
In thermodynamic interpretations, one separates:
\[
\frac{d}{dt}S(\rho) = \Pi(t) - \Phi(t),
\]
where:
\begin{itemize}
\item $\Pi(t)\ge 0$ is \textbf{entropy production} (irreversible),
\item $\Phi(t)$ is \textbf{entropy flux} to/from environment.
\end{itemize}

\medskip
Spohn's inequality provides a rigorous non-negativity statement for $\Pi(t)$
in Markovian dynamics with a stationary state $\rho_\ast$.
\end{frame}
